// Власна функція для порівняння частини C-рядка (масиву char) з іншим рядком.
// Повертає: 0 (якщо однакові), -1 (якщо перший менший), 1 (якщо перший більший)
int custom_compare(const char* str, size_t pos, size_t len, const char* s) {
    // 1. Рахуємо довжину основного C-рядка вручну, оскільки заборонено (або небажано) використовувати strlen
    size_t str_len = 0;
    while (str[str_len] != '\0') { // '\0' - символ кінця рядка
        str_len++;
    }

    // 2. Захист від виходу за межі рядка. Якщо позиція початку більша за довжину рядка,
    // ми просто зсуваємо її в самий кінець.
    if (pos > str_len) {
        pos = str_len;
    }

    // 3. Обчислюємо реальну довжину підрядка для порівняння.
    // Якщо від позиції pos до кінця рядка залишилося менше символів, ніж ми просимо (len),
    // беремо лише той залишок (str_len - pos). Інакше беремо повну довжину (len).
    size_t rlen = (str_len - pos < len) ? (str_len - pos) : len;

    size_t i = 0;
    // 4. Посимвольне порівняння 
    while (i < rlen && s[i] != '\0') {
        if (str[pos + i] != s[i]) {
            return (str[pos + i] < s[i]) ? -1 : 1;
        }
        i++;
    }

    // 5. Якщо всі перевірені символи збіглися, перевіряємо, хто закінчився першим
    if (i < rlen) return 1;      // Основний підрядок має ще символи (він більший)
    if (s[i] != '\0') return -1; // Рядок порівняння має ще символи (він більший)
    return 0;                    // Рядки абсолютно ідентичні
}

void run_task1() {
    std::cout << "\n--- Виконання Завдання 1 (Варіант 45) ---\n";
    // Очищаємо буфер введення, щоб уникнути конфліктів, якщо раніше використовувався cin
    std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');

    std::cout << "Введіть основний рядок: ";
    std::string std_str;
    std::getline(std::cin, std_str); // Читаємо весь рядок з пробілами

    // Демонстрація роботи з масивами символів 
    char char_arr[256];
    size_t idx = 0;
    // Вручну копіюємо символи з std::string у char масив
    for (; idx < std_str.length() && idx < 255; ++idx) {
        char_arr[idx] = std_str[idx];
    }
    char_arr[idx] = '\0'; // Обов'язково додаємо нуль-термінатор у кінці!

    size_t pos, len;
    std::cout << "Введіть позицію початку порівняння : ";
    std::cin >> pos;
    std::cout << "Введіть довжину підрядка : ";
    std::cin >> len;

    // Оскільки перед цим був cin >>, у буфері залишився символ Enter ('\n').
    // Його треба прибрати перед наступним getline, інакше він прочитає порожній рядок.
    std::cin.ignore();

    std::cout << "Введіть рядок для порівняння : ";
    std::string compare_str;
    std::getline(std::cin, compare_str);

    // Виклик власної функції та стандартного методу C++ для порівняння
    // c_str() конвертує std::string у const char* для нашої функції
    int res_custom = custom_compare(char_arr, pos, len, compare_str.c_str());
    int res_std = std_str.compare(pos, len, compare_str.c_str());

    // Метод compare може повертати будь-які від'ємні чи додатні числа.
    // Нормалізуємо їх до -1, 0, 1 для коректного порівняння з нашою функцією.
    if (res_std < 0) res_std = -1;
    if (res_std > 0) res_std = 1;

    std::cout << "\nРезультат власної функції: " << res_custom << "\n";
    std::cout << "Результат стандартного методу: " << res_std << "\n";

    if (res_custom == res_std) {
        std::cout << ">> Успіх! Результати абсолютно однакові. \n";
    }
    else {
        std::cout << ">> Помилка! Результати відрізняються. \n";
    }
}

// Перевіряє, чи складається рядок виключно з цифр і чи не дорівнює нулю
bool is_valid_positive_integer(const std::string& str) {
    if (str.empty()) return false; // Порожній рядок - не число

    // Проходимось по кожному символу рядка
    for (char c : str) {
        // Якщо символ не лежить у діапазоні від '0' до '9' — це не цифра
        if (c < '0' || c > '9') return false;
    }

    // Рядки типу "000" теж пройдуть першу перевірку. 
    // Тому рахуємо суму: якщо вона > 0, отже, є хоча б одна значуща цифра.
    int sum = 0;
    for (char c : str) {
        sum += (c - '0'); // Віднімаємо код '0', щоб отримати числове значення (наприклад, '5' - '0' = 5)
    }
    return sum > 0;
}

// Функція для розрахунку суми цифр
int calculate_digit_sum(const std::string& str) {
    int sum = 0;
    for (char c : str) {
        if (c >= '0' && c <= '9') {
            sum += (c - '0'); // Конвертація символу в число і додавання до суми
        }
    }
    return sum;
}

// Створює тестовий файл, щоб не вводити дані щоразу вручну
void create_sample_file(const std::string& filename) {
    std::ofstream out(filename); // ofstream - потік для ЗАПИСУ у файл
    if (!out) return; // Якщо файл не вдалося створити, виходимо

    // Записуємо тестові випадки 
    out << "12345\n";      // Валідний (Сума: 15)
    out << "9081\n";       // Валідний (Сума: 18)
    out << "0\n";          // Невалідний (Нуль не є позитивним числом)
    out << "7\n";          // Валідний (Сума: 7)
    out << "9999\n";       // Валідний (Сума: 36)
    out << "100a23\n";     // Невалідний (Містить літеру 'a')
    out << "000025\n";     // Валідний (Сума: 7, провідні нулі допускаються)
    out << "-543\n";       // Невалідний (Від'ємне число містить символ '-')
    out << "1111111111\n"; // Валідний (Сума: 10)
    out << "hello_world\n";// Невалідний рядок
    out.close();           // Обов'язково закриваємо файл після роботи
}

// Зчитує дані з одного файлу, обробляє і записує в інший
void process_file(const std::string& input_filename, const std::string& output_filename) {
    std::ifstream in(input_filename);   // ifstream - потік для ЧИТАННЯ
    std::ofstream out(output_filename); // ofstream - потік для ЗАПИСУ

    // Перевірка, чи успішно відкрилися обидва файли
    if (!in.is_open() || !out.is_open()) {
        std::cerr << "Помилка відкриття файлів! Перевірте правильність імен.\n";
        return;
    }

    std::string line;
    // Цикл читає файл по одному рядку до самого кінця
    while (std::getline(in, line)) {
        if (is_valid_positive_integer(line)) {
            int sum = calculate_digit_sum(line);
            out << "Рядок: " << line << " -> Сума цифр: " << sum << "\n";
        }
        else {
            out << "Рядок: " << line << " -> Помилка: не є цілим позитивним числом\n";
        }
    }

    in.close();
    out.close();
}
