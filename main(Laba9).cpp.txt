#include <iostream>
#include <clocale>
#include <limits>
#include <Windows.h> // Необхідно для керування кодовими сторінками консолі Windows
#include "string_tasks.h" // Підключення заголовного файлу з вашими користувацькими функціями 

// Оголошуємо прототип функції меню. 
// Це потрібно для того, щоб функція main() знала про її існування, 
// хоча сама реалізація (тіло функції) знаходиться нижче по коду.
void show_menu();

int main() {
    // --- НАЛАШТУВАННЯ МОВИ ТА КОДУВАННЯ ---

    // Встановлюємо українську локаль для стандартних функцій C
    std::setlocale(LC_ALL, "Ukrainian");

    // Специфічні налаштування для коректного відображення кирилиці саме в консолі Windows
    SetConsoleCP(1251);       // Перемикає введення з клавіатури на кодування Windows-1251
    SetConsoleOutputCP(1251); // Перемикає виведення на екран на кодування Windows-1251

    // Запуск головного меню програми, де відбувається вся взаємодія з користувачем
    show_menu();

    // Повертаємо 0, що сигналізує операційній системі про успішне завершення програми
    return 0;
}

// --- ЛОГІКА ІНТЕРАКТИВНОГО МЕНЮ ---
void show_menu() {
    int choice; // Змінна для зберігання пункту меню, який обере користувач

    // Використовуємо цикл do-while, щоб меню гарантовано показалося хоча б один раз
    do {
        // Виведення інтерфейсу меню на екран
        std::cout << "\n================= ГОЛОВНЕ МЕНЮ =================\n";
        std::cout << "1. Виконати Завдання 1 (Варіант 45) \n";
        std::cout << "2. Виконати Завдання 2 (Варіант 23) \n";
        std::cout << "0. Вийти з програми\n";
        std::cout << "================================================\n";
        std::cout << "Оберіть дію: ";

        // Захист від некоректного введення 
        // Якщо користувач введе літеру або текст замість числа, std::cin не зможе зчитати дані 
        // у змінну типу int і перейде у стан помилки. Цикл while перехоплює цю ситуацію.
        while (!(std::cin >> choice)) {
            std::cout << "Некоректне введення! Будь ласка, введіть число: ";

            std::cin.clear(); 

            // Очищаємо буфер введення від некоректних символів аж до натискання Enter ('\n').
            // Це не дасть програмі зациклитися на одному й тому ж неправильному вводі.
            std::cin.ignore((std::numeric_limits<std::streamsize>::max)(), '\n');
        }

        // Обробка коректно введеного числа за допомогою конструкції switch
        switch (choice) {
        case 1:
            run_task1(); // Виклик функції для першого завдання 
            break;       // Використовуємо break, щоб зупинити виконання switch після запуску case
        case 2:
            run_task2(); // Виклик функції для другого завдання
            break;
        case 0:
            std::cout << "Програма завершила роботу.\n"; // Повідомлення при виході
            break;
        default:
            // default спрацьовує, якщо введено число, для якого немає case (наприклад, 3 або 99)
            std::cout << "Такого пункту меню немає. Спробуйте ще раз.\n";
        }
    } while (choice != 0); // Програма повторюватиме цикл, доки користувач не введе 0
}
